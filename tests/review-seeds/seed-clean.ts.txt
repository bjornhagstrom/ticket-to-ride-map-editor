// Rounds a position to a thousandth of a unit so that a map file does not carry noise from the pointer.
export const roundPosition = (value: number): number => Math.round(value * 1000) / 1000;

/** The last element of a list that is known to have one, as the callers guarantee. */
export const lastOf = <T,>(items: T[]): T => items[items.length - 1];
